⚠️ Apps Script must use JSON body. Your doPost must read e.postData.contents not FormData.
function doPost(e) {
  var p = JSON.parse(e.postData.contents);
  var blob = Utilities.newBlob(Utilities.base64Decode(p.data), p.mimeType, p.fileName);
  var file = DriveApp.getFolderById('FOLDER_ID').createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return ContentService.createTextOutput(JSON.stringify({id:file.getId()})).setMimeType(ContentService.MimeType.JSON);
}
